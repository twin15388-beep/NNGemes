cmake_minimum_required(VERSION 3.20)
project(nzl_studio LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(nzl_core STATIC src/chain.cpp src/esp_pipeline.cpp src/skeleton.cpp)
target_include_directories(nzl_core PUBLIC include)
if(MSVC)
  target_compile_options(nzl_core PRIVATE /W4 /permissive-)
else()
  target_compile_options(nzl_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(nzl_tests tests/test_chain.cpp)
target_link_libraries(nzl_tests PRIVATE nzl_core)
add_test(NAME nzl_tests COMMAND nzl_tests)
add_executable(nzl_pipeline_tests tests/test_pipeline.cpp)
target_link_libraries(nzl_pipeline_tests PRIVATE nzl_core)
add_test(NAME nzl_pipeline_tests COMMAND nzl_pipeline_tests)
add_executable(nzl_profile_tests tests/test_profile.cpp)
target_link_libraries(nzl_profile_tests PRIVATE nzl_core)
add_test(NAME nzl_profile_tests COMMAND nzl_profile_tests)
add_executable(nzl_skeleton_tests tests/test_skeleton.cpp)
target_link_libraries(nzl_skeleton_tests PRIVATE nzl_core)
add_test(NAME nzl_skeleton_tests COMMAND nzl_skeleton_tests)

# Driver wire protocol: layout ABI, address validation, batch checks, the
# kernel scan state machine and the PEB module walk. Host-side only; the
# same headers are compiled into driver.sys by MSVC+WDK.
add_executable(nzl_protocol_tests tests/test_protocol.cpp)
target_include_directories(nzl_protocol_tests PRIVATE include)
if(MSVC)
  target_compile_options(nzl_protocol_tests PRIVATE /W4 /permissive-)
else()
  target_compile_options(nzl_protocol_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME nzl_protocol_tests COMMAND nzl_protocol_tests)
if(WIN32)
  add_executable(nzl_probe src/probe_windows.cpp client/sc_client.cpp)
  target_include_directories(nzl_probe PRIVATE client)
  target_link_libraries(nzl_probe PRIVATE nzl_core winmm advapi32 bcrypt)
  add_executable(nzl_studio src/main_windows.cpp src/driver_memory_windows.cpp client/sc_client.cpp)
  target_include_directories(nzl_studio PRIVATE client)
  target_link_libraries(nzl_studio PRIVATE nzl_core winmm bcrypt)
  add_executable(nzl_overlay src/overlay_windows.cpp src/driver_memory_windows.cpp client/sc_client.cpp)
  target_include_directories(nzl_overlay PRIVATE client)
  target_link_libraries(nzl_overlay PRIVATE nzl_core winmm bcrypt)
endif()
